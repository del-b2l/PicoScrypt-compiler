--ast:

python3 main.py examples/manor.pico --ast

AST

WorldNode('Manor')
  FlagNode('west_wing_unlocked' = True)
  FlagNode('vault_cracked' = False)
  PlayerNode
    InvNode('crowbar')
  ItemNode('crowbar')
    ExamineAction('A heavy iron crowbar. Useful for prying things open.')
  ItemNode('portrait')
    ExamineAction('A stern portrait. Its eyes seem to follow you.')
  RoomNode('foyer')
    ItemRef('portrait')
    ExitNode(west -> 'west_wing')
      ConditionFlag('west_wing_unlocked')
  RoomNode('west_wing')
    OnEnterBlock
      SayStmt('A cold draft sweeps through the darkened corridor.')
    ExitNode(south -> 'vault')
      ConditionFlag('vault_cracked')
  RoomNode('vault')
    OnEnterBlock
      WinStmt('You crack the vault. Inside: the missing will.')
  PuzzleNode(unlock='vault_cracked')
    ConditionInv(has 'crowbar')

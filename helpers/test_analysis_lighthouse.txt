--ast:

python3 main.py examples/lighthouse.pico --ast

AST

WorldNode('Lighthouse')
  FlagNode('lamp_lit' = False)
  PlayerNode
    InvNode('oil')
  ItemNode('oil')
    ExamineAction('A flask of lamp oil.')
    UseAction
      SayStmt('You pour the oil into the lamp reservoir.')
  RoomNode('shore')
    ExitNode(north -> 'tower')
      ConditionFlag('lamp_lit')
  RoomNode('tower')
    OnEnterBlock
      WinStmt('The beam sweeps out to sea. The ship is saved!')
  NPCNode('keeper')
    DialogueNode
      DialogueBranch(if)
        ConditionFlag('lamp_lit')
        then:
          SayStmt('The lamp burns bright. Well done.')
        else:
          SayStmt('The lamp is dark. Find oil to light it.')
  PuzzleNode(unlock='lamp_lit')
    ConditionInv(has 'oil')

RAW TAC

000: ROOM_TOWER_ENTER:
001: PRINT "The beam sweeps out to sea. The ship is saved!"
002: HALT
003: ROOM_TOWER_END:
004: t1 = inv_has(oil)
005: if t1 goto PUZZLE_LAMP_LIT_UNLOCK_1
006: goto PUZZLE_LAMP_LIT_END_1
007: PUZZLE_LAMP_LIT_UNLOCK_1:
008: lamp_lit = true
009: PUZZLE_LAMP_LIT_END_1:

--ast:

python3 main.py examples/crypt.pico --ast

AST

WorldNode('Crypt')
  FlagNode('door_open' = False)
  PlayerNode
    InvNode('key')
  ItemNode('key')
    ExamineAction('An old iron key.')
  RoomNode('entrance')
    ItemRef('gem')
    ExitNode(north -> 'chamber')
      ConditionFlag('door_open')
  ItemNode('gem')
    UseAction
      SayStmt('The gem clicks into place.')
  RoomNode('chamber')
    OnEnterBlock
      WinStmt('You escaped!')
  PuzzleNode(unlock='door_open')
    ConditionInv(has 'key')

--ast:

python3 main.py examples/shipwreck.pico --ast

AST

WorldNode('Shipwreck')
  FlagNode('hull_patched' = False)
  FlagNode('anchor_raised' = False)
  PlayerNode
    InvNode('rope')
  ItemNode('rope')
    ExamineAction('A length of knotted rope.')
    UseAction
      SayStmt('You lash the rope around the capstan.')
  ItemNode('patch_kit')
    ExamineAction('Tar and canvas for sealing hull breaches.')
    UseAction
      SayStmt('You press the patch over the crack. It holds.')
  RoomNode('lower_deck')
    ItemRef('patch_kit')
    OnEnterBlock
      SayStmt('Water sloshes around your ankles.')
    ExitNode(north -> 'upper_deck')
      ConditionFlag('hull_patched')
  RoomNode('upper_deck')
    OnEnterBlock
      SayStmt('Salt wind tears at your coat.')
    ExitNode(north -> 'helm')
      ConditionFlag('anchor_raised')
  RoomNode('helm')
    OnEnterBlock
      WinStmt('You steer the battered ship into harbour. You survived.')
  NPCNode('bosun')
    DialogueNode
      DialogueBranch(if)
        ConditionFlag('hull_patched')
        then:
          SayStmt('Good work below. Now raise the anchor once you exit the lower deck.')
        else:
          SayStmt('Patch the hull first or we sink where we stand.')
  PuzzleNode(unlock='hull_patched')
    ConditionInv(has 'patch_kit')
  PuzzleNode(unlock='anchor_raised')
    ConditionInv(has 'rope')

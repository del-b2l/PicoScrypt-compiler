--ast:

python3 main.py examples/tomb.pico --ast

AST

WorldNode('Tomb')
  FlagNode('sarcophagus_open' = False)
  PlayerNode
    InvNode('torch')
    InvNode('scarab')
  ItemNode('torch')
    ExamineAction('A flickering wooden torch.')
    UseAction
      LoseStmt('The torch goes out. Something ancient stirs in the darkness.')
  ItemNode('scarab')
    ExamineAction('A golden scarab amulet.')
    UseAction
      SayStmt('The scarab glows and fits into the door.')
  RoomNode('antechamber')
    ExitNode(east -> 'burial_chamber')
      ConditionFlag('sarcophagus_open')
  RoomNode('burial_chamber')
    OnEnterBlock
      WinStmt('The pharaoh sleeps peacefully. The treasure is yours!')
  PuzzleNode(unlock='sarcophagus_open')
    ConditionInv(has 'scarab')
